    function renderTemplates() {

      const wrap = $("templateList");

      const heading = $("templateListHeading");

      wrap.innerHTML = "";


      const query = state.search.trim().toLowerCase();

      const searchPanel = $("searchPanel");

      if (searchPanel && query) searchPanel.open = true;

      const relevant = (query

        ? templates.filter(t => [t.title, t.bestFor, t.why, t.sourcePath, t.category].join(" ").toLowerCase().includes(query))

        : templates.filter(t => t.category === state.category)

      ).sort((a, b) => {

        if (a.category !== b.category) return categories.findIndex(c => c.id === a.category) - categories.findIndex(c => c.id === b.category);

        return templateSortValue(a) - templateSortValue(b);

      });


      heading.textContent = query ? `Search results (${relevant.length})` : `Available prompts (${relevant.length})`;


      if (!query && relevant.length) {

        const guidance = document.createElement("p");

        guidance.className = "mb-3 text-xs leading-5 text-slate-500";

        guidance.textContent = "Choose one prompt below. You can switch later.";

        wrap.appendChild(guidance);

      }


      if (!relevant.length) {

        const empty = document.createElement("div");

        empty.className = "rounded-2xl border border-slate-200 bg-slate-50 p-4 text-sm leading-6 text-slate-600";

        empty.textContent = "No prompts match that search. Try a broader term like feedback, hook, scaffold, rubric, or assessment.";

        wrap.appendChild(empty);

        return;

      }


      relevant.forEach(template => {

        const button = document.createElement("button");

        button.type = "button";

        button.className = "template-button w-full rounded-2xl border border-slate-200 p-4 text-left hover:bg-slate-50";

        button.setAttribute("aria-pressed", String(state.templateChosen && template.id === state.templateId));

        button.innerHTML = `<span class="block text-sm font-semibold">${template.title}</span><span class="mt-1 block text-xs leading-5 text-slate-500">${template.bestFor}</span>`;

        button.addEventListener("click", () => {

          state.templateId = template.id;

          state.category = template.category;

          state.categoryChosen = true;

          state.templateChosen = true;

          state.values = {};

          render();

          state.activeStep = 3;

          updateStepHighlights();

          scrollToStep(3);

        });

        wrap.appendChild(button);

      });

    }

    function buildSelectedPrompt() {

      const template = selectedTemplate();

      if (!template) return "Choose a category and template to begin building your prompt.";


      let text = template.text;

      template.fields.forEach(field => {

        text = text.replaceAll(`{{${field.id}}}`, getValue(field.id, field.label));

      });


      const desiredFormat = $("desiredFormat").value.trim();

      const toneStyle = $("toneStyle").value.trim();

      const additionalConstraints = $("additionalConstraints").value.trim();


      const optional = [];

      if (desiredFormat) optional.push(`Desired format: ${desiredFormat}`);

      if (toneStyle) optional.push(`Tone or style: ${toneStyle}`);

      if (additionalConstraints) optional.push(`Additional constraints: ${additionalConstraints}`);


      if (optional.length) {

        text += `\n\nOPTIONAL CONSTRAINTS / NOTES\n${optional.map(item => `- ${item}`).join("\n")}`;

      }


      return `TASK: ${template.title}


${text}`;

    }


    function buildFullPrompt() {

      return `${buildContextSection()}


---


${buildSelectedPrompt()}`;

    }


    function buildIterationPrompt() {

      const template = selectedTemplate();

      const defaultFollowUp = "Revise this output to be more specific, less generic, and more directly aligned with the class context I provided. Do not add unnecessary complexity. Identify one assumption you made and one detail I could add to improve the result.";

      return template?.followUp || defaultFollowUp;

    }


    function findBlankPlaceholders(text) {

      return text.match(/\[Insert [^\]]+\]/g) || [];

    }


    function scrollToElement(element, offset = 24) {

      if (!element) return;

      window.setTimeout(() => {

        const targetTop = element.getBoundingClientRect().top + window.pageYOffset - offset;

        window.scrollTo({ top: Math.max(0, targetTop), behavior: "smooth" });

      }, 120);

    }

    function isStep1Complete() {

      return ["gradeSubject", "topic", "classSize", "classContext"].every(id => $(id).value.trim());

    }


    function isStep2Complete() {

      return state.categoryChosen && state.templateChosen && Boolean(selectedTemplate());

    }


    function isStep3Complete() {

      const template = selectedTemplate();

      if (!template) return false;

      return template.fields.every(field => (state.values[field.id] || "").trim());

    }


    function getActiveStep() {

      if (!isStep1Complete()) return 1;

      if (!isStep2Complete()) return 2;

      if (!isStep3Complete()) return 3;

      return 4;

    }


    function updateStepHighlights() {

      const active = state.activeStep || getActiveStep();

      [1, 2, 3, 4].forEach(step => {

        const panel = $(`step${step}Panel`);

        if (panel) panel.classList.toggle("active-step", step === active);

      });

    }

    async function copyText(kind, button = null) {

      if (kind === "full" && !isStep2Complete()) {

        $("copyStatus").textContent = "Choose a prompt first.";

        window.setTimeout(() => { $("copyStatus").textContent = ""; }, 3000);

        return;

      }


      if (kind === "iteration" && !isStep2Complete()) {

        $("copyStatus").textContent = "Choose a prompt first.";

        window.setTimeout(() => { $("copyStatus").textContent = ""; }, 3000);

        return;

      }

      const options = {

        full: buildFullPrompt(),

        iteration: buildIterationPrompt()

      };


      const text = options[kind] || buildFullPrompt();

      const status = $("copyStatus");


      try {

        await navigator.clipboard.writeText(text);

        const blanks = kind === "full" ? findBlankPlaceholders(text).length : 0;

        status.textContent = blanks ? `Copied to clipboard with ${blanks} placeholder${blanks === 1 ? "" : "s"} still blank.` : "Copied to clipboard.";

        flashCopied(button);

      } catch (error) {

        const fallbackBox = kind === "iteration" ? $("iterationPreview") : $("promptPreview");

        fallbackBox.value = text;

        fallbackBox.focus({ preventScroll: true });

        fallbackBox.select();

        status.textContent = kind === "iteration"

          ? "Clipboard access was blocked. The follow-up prompt is selected so you can copy it manually."

          : "Clipboard access was blocked. The prompt is selected so you can copy it manually.";

      }


      window.setTimeout(() => { status.textContent = ""; }, 3000);

    }


    ["gradeSubject", "topic", "classSize", "classContext", "desiredFormat", "toneStyle", "additionalConstraints"].forEach(id => {

      document.addEventListener("input", (event) => {

        if (event.target && event.target.id === id) updatePreview();

      });

    });


    document.querySelectorAll(".copy-btn").forEach(button => {

      button.addEventListener("click", () => copyText(button.dataset.copy, button));

    });

      scrollToStep(1);

    });

      state.search = event.target.value;

      updateStepHighlights();

    });
